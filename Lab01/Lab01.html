Enfoque analítico esperado: Identificar periodicidad semanal y comparar intensidad por día.

# 1. Extraer la hora del día (0-23)
df['hora_del_dia'] = df['fecha_hora_infraccion'].dt.hour

# 2. Conteo de homicidios por hora
conteo_por_hora = df['hora_del_dia'].value_counts().sort_index()
print("\n--- Conteo de Homicidios por Hora del Día ---")
print(conteo_por_hora)

# 3. Agrupar por franja horaria para ver el patrón (Madrugada, Mañana, Tarde, Noche)
def categorizar_franja_horaria(hora):
    if pd.isna(hora):
        return 'Desconocido'
    elif 0 <= hora < 6:
        return '1 - Madrugada (00:00 - 05:59)'
    elif 6 <= hora < 12:
        return '2 - Mañana (06:00 - 11:59)'
    elif 12 <= hora < 18:
        return '3 - Tarde (12:00 - 17:59)'
    else:
        return '4 - Noche (18:00 - 23:59)'

df['franja_horaria'] = df['hora_del_dia'].apply(categorizar_franja_horaria)
conteo_franja = df['franja_horaria'].value_counts().sort_index()

Pregunta 3: Evolución mensual de homicidios

¿Cómo varían los homicidios a lo largo de los meses del año 2025? ¿Se observa una tendencia creciente, decreciente o estacional?

Enfoque analítico esperado: Detección de tendencias y cambios en el tiempo.

Code 
# 1. Filtrar los datos solo para el año 2025
df_2025 = df[df['fecha_hora_infraccion'].dt.year == 2025].copy()

# 2. Extraer el mes numérico
df_2025['mes_num'] = df_2025['fecha_hora_infraccion'].dt.month.astype(int)

# 3. Mapear a los nombres de los meses en español
meses_map = {
    1: 'Enero', 2: 'Febrero', 3: 'Marzo', 4: 'Abril', 5: 'Mayo', 6: 'Junio',
    7: 'Julio', 8: 'Agosto', 9: 'Septiembre', 10: 'Octubre', 11: 'Noviembre', 12: 'Diciembre'
}
df_2025['mes'] = df_2025['mes_num'].map(meses_map)

# 4. Conteo de homicidios por mes (manteniendo el orden cronológico)
conteo_mensual = df_2025['mes'].value_counts().reindex(list(meses_map.values())).dropna()
print("\n--- Conteo de Homicidios por Mes en 2025 ---")
print(conteo_mensual)

# 5. Visualización gráfica con Altair (Gráfico de líneas para ver tendencias)
# Crear una selección interactiva que siga al cursor
nearest = alt.selection_point(nearest=True, on='mouseover', fields=['Mes'], empty=False)

# Línea base suavizada
line = alt.Chart(df_plot_mes).mark_line(interpolate='monotone', color='#2ca02c', strokeWidth=3).encode(
    x=alt.X('Mes:N', sort=alt.EncodingSortField(field='Mes_Num', order='ascending'), title='Mes del Año 2025', axis=alt.Axis(labelAngle=-45)),
    y=alt.Y('Número de Ocurrencias:Q', scale=alt.Scale(zero=False), title='Número de Homicidios')
)

# Capa invisible para detectar el mouse
selectors = alt.Chart(df_plot_mes).mark_point().encode(
    x=alt.X('Mes:N', sort=alt.EncodingSortField(field='Mes_Num', order='ascending')),
    opacity=alt.value(0)
).add_params(nearest)

# Puntos que aparecen solo cuando pasas el mouse
points = line.mark_point(size=90, filled=True, color='#2ca02c').encode(
    opacity=alt.condition(nearest, alt.value(1), alt.value(0))
)

# Texto que aparece al pasar el mouse
text = line.mark_text(align='left', dx=10, dy=-15, fontSize=13, fontWeight='bold', color='#2ca02c').encode(
    text=alt.condition(nearest, 'Número de Ocurrencias:Q', alt.value(' '))
)

# Línea vertical punteada al pasar el mouse
rules = alt.Chart(df_plot_mes).mark_rule(color='gray', strokeDash=[3,3]).encode(
    x=alt.X('Mes:N', sort=alt.EncodingSortField(field='Mes_Num', order='ascending')),
).transform_filter(nearest)

# Ensamblar las 5 capas
chart_mes = alt.layer(
    line, selectors, points, rules, text
).properties(
    title=alt.TitleParams('Evolución Mensual de Homicidios en 2025 (Pasa el cursor)', fontSize=16),
    width=650, height=350
)

chart_mes.show()

# 4. Visualización Gráfica Mejorada: Pirámide Poblacional Interactiva por Edad Exacta

# Para ver TODAS las edades, agrupamos por la edad exacta (número entero) y sexo.
# Primero limpiamos y filtramos los datos válidos.

df_edades['edad_num'] = df_edades['edad_num'].astype(int)

# Agrupamos en un nuevo DataFrame para no superar el límite de filas de Altair (MaxRowsError)
df_plot_edades = df_edades.groupby(['edad_num', 'sexo']).size().reset_index(name='Número de Ocurrencias')

# Calcular el valor máximo absoluto para unificar las escalas de hombres y mujeres
max_val = df_plot_edades['Número de Ocurrencias'].max()

# Configuramos un selector interactivo para resaltar al pasar el cursor (hover)
hover = alt.selection_point(on='mouseover', nearest=True, fields=['edad_num'], empty=True)

# Gráfico base
base = alt.Chart(df_plot_edades).properties(width=300, height=500)

# Lado izquierdo (Hombres)
left = base.transform_filter(
    alt.datum.sexo == 'HOMBRE'
).encode(
    y=alt.Y('edad_num:O', axis=None, sort='descending'), 
    x=alt.X('Número de Ocurrencias:Q', title='Hombres', sort=alt.SortOrder('descending'), scale=alt.Scale(domain=[0, max_val])),
    # Color más claro por defecto, oscuro al pasar el cursor
    color=alt.condition(hover, alt.value('#1f77b4'), alt.value('#a1c9f4')), 
    tooltip=[alt.Tooltip('edad_num:O', title='Edad Exacta'), alt.Tooltip('Número de Ocurrencias:Q', title='Homicidios (Hombres)')]
).mark_bar(height=3).add_params(
    hover
).properties(title='Hombres')

# Eje central (Texto de Edades)
# Mostramos texto solo para las edades múltiplo de 5 para no saturar el eje, manteniendo el espacio de las demás
middle = base.encode(
    y=alt.Y('edad_num:O', axis=None, sort='descending'),
    text=alt.condition("datum.edad_num % 5 == 0", alt.Text('edad_num:O'), alt.value(''))
).mark_text(fontWeight='bold', fontSize=10).properties(width=40, title='Edad')

# Lado derecho (Mujeres)
right = base.transform_filter(
    alt.datum.sexo == 'MUJER'
).encode(
    y=alt.Y('edad_num:O', axis=None, sort='descending'),
    x=alt.X('Número de Ocurrencias:Q', title='Mujeres', scale=alt.Scale(domain=[0, max_val])),
    # Color más claro por defecto, oscuro al pasar el cursor
    color=alt.condition(hover, alt.value('#e377c2'), alt.value('#fbcfe8')), 
    tooltip=[alt.Tooltip('edad_num:O', title='Edad Exacta'), alt.Tooltip('Número de Ocurrencias:Q', title='Homicidios (Mujeres)')]
).mark_bar(height=3).add_params(
    hover
).properties(title='Mujeres')

# Ensamblamos la pirámide concatenando de forma horizontal
chart_demografia = alt.concat(left, middle, right, spacing=2).properties(
    title=alt.TitleParams('Pirámide Demográfica Interactiva por Edad Exacta y Sexo', fontSize=18, anchor='middle')
