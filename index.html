grafico = alt.Chart(df).mark_boxplot(
    size=40,          # Grosor de las cajas
    outliers=True     # Mostrar puntos atípicos
).encode(
    # Eje X: Especie
    x=alt.X('species:N', title=None, axis=alt.Axis(labels=False, ticks=False)),
    
    # Eje Y: Masa corporal.
    y=alt.Y('body_mass_g:Q', 
            title='Masa Corporal (gramos)', 
            scale=alt.Scale(zero=False)),
    
    # Color: Asignamos un color por especie
    color=alt.Color('species:N', 
                    title='Especie', 
                    scale=alt.Scale(scheme='set2')),
    
    # Facetado (Small Multiples): Una columna por cada Isla
    column=alt.Column('island:N', 
                      title='Isla', 
                      header=alt.Header(labelOrient='bottom', titleOrient='bottom', labelFontSize=12))
    
).properties(
    width=150,  # Ancho de cada subgráfico
    height=350, # Alto de los gráficos
    title="Distribución de Masa Corporal de Pingüinos por Especie e Isla"
).configure_title(
    fontSize=18,
    anchor='start',
    offset=20
).configure_facet(
    spacing=15 # Espacio entre los paneles
).configure_view(
    stroke=None # Quitar el borde exterior de los paneles
)
grafico.show()

# Crear el Mapa de Calor base
heatmap = alt.Chart(df).mark_rect().encode(
    x=alt.X('island:N', title='Isla', axis=alt.Axis(labelAngle=0, labelFontSize=12)),
    y=alt.Y('species:N', title='Especie', axis=alt.Axis(labelFontSize=12)),
    color=alt.Color('count():Q', 
                    title='Cant. de Pingüinos', 
                    scale=alt.Scale(scheme='blues')), # Paleta de azules
    tooltip=['island', 'species', 'count()']
).properties(
    width=400,
    height=400,
    title='Presencia de Especies por Isla (Mapa de Calor)'
)

# Añadir las etiquetas de texto sobre los rectángulos
text = heatmap.mark_text(baseline='middle', fontSize=14, fontWeight='bold').encode(
    text='count():Q',
    color=alt.condition(
        alt.datum['count()'] > 40,
        alt.value('white'),
        alt.value('black')
    )
)

# Superponer texto y mapa de calor
grafico_ubicacion = heatmap + text

grafico_ubicacion.show()

df_sex = df.dropna(subset=['culmen_depth_mm', 'species', 'sex'])
df_sex = df_sex[df_sex['sex'].isin(['MALE', 'FEMALE'])]

# Grafico de densidad superpuesto
grafico_sexos = alt.Chart(df_sex).transform_density(
    density='culmen_depth_mm',
    groupby=['sex', 'species'],
    as_=['culmen_depth_mm', 'density']
).mark_area(
    opacity=0.6 # transparencia para ver la superposición
).encode(
    x=alt.X('culmen_depth_mm:Q', title='Profundidad del Pico (mm)'),
    y=alt.Y('density:Q', title='Densidad', axis=alt.Axis(labels=True, ticks=True), stack=None), # Oculto el eje Y 
    color=alt.Color('sex:N', 
                    title='Sexo', 
                    scale=alt.Scale(domain=['FEMALE', 'MALE'], range=['#e15759', '#4e79a7'])), # Rojo para hembras, Azul para machos
    column=alt.Column('species:N', title='Especie')
).properties(
    width=200,
    height=250,
    title='Dimorfismo Sexual: Profundidad del Pico (Machos vs. Hembras)'
).configure_title(
    fontSize=16,
    anchor='start'
)

grafico_sexos.show()

Justificacion del tipo de gráfico
- Para comparar la distribución completa de solo dos categorías (Machos vs. Hembras) sobre una variable continua, la técnica más estética y efectiva es un Gráfico de Densidad Superpuesto con opacidad.

  • ¿Por qué no un Histograma? Los histogramas superpuestos pueden crear ruido visual donde las barras se cruzan, dificultando ver qué color está detrás.

  • ¿Por qué Densidad? El KDE suaviza los datos mostrando la “forma” de la distribución como una curva fluida.

Explicación del gráfico
Dimorfismo Sexual: En las tres especies, existe una diferencia notable en la profundidad del pico entre machos y hembras.

Desplazamiento de las Curvas: En cada panel (especie), la curva azul (Machos) está desplazada hacia la derecha en comparación con la curva roja (Hembras). Esto significa que los machos, en promedio, tienen picos más profundos (gruesos) que las hembras.

Zona de Solapamiento: Aunque los machos tienen picos más grandes, el gráfico de densidad con opacidad nos muestra que existe un área donde el azul y el rojo se solapan. Esto indica que no es una regla absoluta: existen machos con picos pequeños y hembras con picos grandes, pero la tendencia general de la población es innegable.

grafico_proporcion = alt.Chart(df_sex).mark_bar(cornerRadiusEnd=3).encode(
    # Eje X: Ahora usamos la Especie para el eje principal dentro de cada panel
    x=alt.X('species:N', title='Especie', axis=alt.Axis(labelAngle=0, labelFontSize=12)),
    
    # xOffset agrupa las barras de 'sex' una al lado de otra por cada 'species'
    xOffset='sex:N', 
    
    # Eje Y: Conteo absoluto de pingüinos
    y=alt.Y('count():Q', 
            title='Cantidad Exacta de Pingüinos', 
            axis=alt.Axis(labelFontSize=12)),
    
    color=alt.Color('sex:N', 
                    title='Sexo', 
                    scale=alt.Scale(domain=['FEMALE', 'MALE'], range=['#e15759', '#4e79a7'])),
                    
    # Facetado: Separamos el gráfico completo en columnas por 'island'
    column=alt.Column('island:N', title='Isla', header=alt.Header(labelFontSize=14, titleFontSize=14)),
    
    # Tooltip
    tooltip=[alt.Tooltip('island:N', title='Isla'), 
             alt.Tooltip('species:N', title='Especies'),
             alt.Tooltip('sex:N', title='Sexo'), 
             alt.Tooltip('count():Q', title='Cantidad')]
).properties(
    width=150,  # Ancho por cada panel (isla)
    height=300,
    title='Cantidad de Machos vs Hembras por Especie e Isla'
).configure_title(
    fontSize=18,
    anchor='start',
    offset=15
)

grafico_proporcion.show()
